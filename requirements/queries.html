GR01: ¿Cuáles son los centros de gestión de residuos registrados y qué tipo de residuo gestionan? 
# GR01: ¿Cuáles son los centros de gestión de residuos registrados y qué tipo de residuo gestionan?

PREFIX waste: <https://w3id.org/def/medioambiente/residuos#>
PREFIX schema: <https://schema.org/>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>

SELECT ?facility ?legalName ?registrationNumber ?wasteType ?wasteTypeLabel
WHERE {
  ?facility a waste:WasteFacility ;
            waste:registrationNumber ?registrationNumber .

  OPTIONAL { ?facility schema:legalName ?legalName }

  OPTIONAL {
    ?facility waste:managesWasteType ?wasteType .
    OPTIONAL { ?wasteType skos:prefLabel ?wasteTypeLabel }
  }
}
ORDER BY ?facility

GR02: ¿Qué contenedores existen y qué tipo de residuo aceptan? 
# GR02: ¿Qué contenedores existen y qué tipo de residuo aceptan?

PREFIX waste: <https://w3id.org/def/medioambiente/residuos#>
PREFIX schema: <https://schema.org/>
PREFIX skos: <http://www.w3.org/2004/02/skos/core#>

SELECT ?container ?model ?wasteType ?wasteTypeLabel
WHERE {
  ?container a waste:WasteContainer .

  OPTIONAL { ?container schema:model ?model }

  OPTIONAL {
    ?container waste:acceptsWasteType ?wasteType .
    OPTIONAL { ?wasteType skos:prefLabel ?wasteTypeLabel }
  }
}
ORDER BY ?container

GR03: ¿Cuáles son los puntos limpios disponibles y cuáles son fijos o móviles? 
# GR03: ¿Cuáles son los puntos limpios disponibles y cuáles son fijos o móviles?

PREFIX waste: <https://w3id.org/def/medioambiente/residuos#>
PREFIX schema: <https://schema.org/>

SELECT ?collectionPoint ?displayName ?type
WHERE {
  {
    ?collectionPoint a waste:PermanentCollectionPoint .
    BIND("Punto limpio fijo" AS ?type)
  }
  UNION
  {
    ?collectionPoint a waste:MobileCollectionPoint .
    BIND("Punto limpio móvil" AS ?type)
  }

  OPTIONAL { ?collectionPoint schema:name ?name }

  BIND(COALESCE(STR(?name), STRAFTER(STR(?collectionPoint), "/residuos/")) AS ?displayName)
}
ORDER BY LCASE(STR(?displayName)) ?collectionPoint

GR04: ¿Dónde se localizan los recursos asociados a la gestión de residuos? 
# GR04: ¿Dónde se localizan los recursos asociados a la gestión de residuos?

PREFIX waste: <https://w3id.org/def/medioambiente/residuos#>
PREFIX geo: <http://www.opengis.net/ont/geosparql#>

SELECT ?resource ?resourceType ?geometry ?wkt
WHERE {
  VALUES ?resourceType {
    waste:WasteFacility
    waste:WasteContainer
    waste:PermanentCollectionPoint
    waste:MobileCollectionPoint
  }

  ?resource a ?resourceType ;
            geo:hasGeometry ?geometry .

  ?geometry geo:asWKT ?wkt .
}
ORDER BY ?resourceType ?resource

GR05: ¿A qué barrio, distrito o municipio pertenece cada recurso de gestión de residuos? 
# GR05: ¿A qué barrio, distrito o municipio pertenece cada recurso de gestión de residuos?

PREFIX waste: <https://w3id.org/def/medioambiente/residuos#>
PREFIX esadm: <https://vocab.linkeddata.es/datosabiertos/def/sector-publico/territorio#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?resource ?territorialEntity ?territorialType ?displayLabel
WHERE {
  ?resource waste:belongsTo ?territorialEntity .

  ?territorialEntity a ?territorialType .

  VALUES ?territorialType {
    esadm:Barrio
    esadm:Distrito
    esadm:Municipio
  }

  OPTIONAL { ?territorialEntity rdfs:label ?label }

  BIND(COALESCE(STR(?label), STRAFTER(STR(?territorialEntity), "/residuos/")) AS ?displayLabel)
}
ORDER BY ?resource ?territorialType ?displayLabel
